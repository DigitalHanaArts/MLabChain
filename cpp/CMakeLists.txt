cmake_minimum_required(VERSION 3.20)
if(POLICY CMP0167)
  cmake_policy(SET CMP0167 NEW)
endif()
project(mera_core VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(OpenSSL REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(Boost REQUIRED)
find_package(Threads REQUIRED)
add_executable(mera_core mera_core.cpp)
target_link_libraries(mera_core PRIVATE OpenSSL::Crypto SQLite::SQLite3 Boost::boost Threads::Threads)
if (MSVC)
  target_compile_options(mera_core PRIVATE /W4 /EHsc)
else()
  target_compile_options(mera_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME mera_self_test COMMAND mera_core self-test --data-dir ${CMAKE_BINARY_DIR}/selftest-data)

install(TARGETS mera_core RUNTIME DESTINATION bin)
